# LinguaFlow — 各板块独立保存代码规范

> 面向 AI 编程助手（Cursor / Claude / Copilot）的开发规范文档。
> 解决：每个板块练习结束后，数据怎么落盘、怎么串联 XP / SRS / 每日任务 / streak。
> 技术栈：React Native + SQLite（expo-sqlite），无后端，TypeScript strict。

---

## 一、设计原则

| 原则 | 说明 |
|------|------|
| 单一保存入口 | 每个板块只有一个 `saveSession()` 函数，所有落盘逻辑集中于此 |
| 事务包裹 | 一次保存 = 一个 `StorageService.transaction()`，要么全成要么全回滚 |
| 级联调用标准化 | 每次保存后固定触发：XP → SRS enroll → streak → 每日任务 → 成就检查 |
| 不跨板块写数据 | 词汇板块只写词汇相关表，不碰听力表 |
| 幂等 | 同一 session 重复调用 `saveSession()` 不重复计 XP / 不重复写流水 |

## 二、公共保存管线（所有板块共用）

```typescript
// services/SessionSavePipeline.ts

export interface SaveContext {
  userId: string;
  board: 'vocab' | 'listening' | 'reading' | 'speaking' | 'writing' | 'review';
  sessionId: string;
  startedAt: string;
  completedAt: string;
  totalItems: number;
  correctCount: number;
  wrongCount: number;
  skippedCount: number;
  accuracy: number;           // correctCount / (correctCount + wrongCount)
  timeSpentSec: number;
  level: string;             // 本次练习的 CEFR 锚点
  responses: SessionResponse[];  // 每题明细
  metadata?: Record<string, any>; // 板块特有数据
}

export interface SessionResponse {
  itemId: string;
  itemType: string;
  correct: boolean;
  rating?: 1 | 2 | 3 | 4;   // 复习板块用
  responseTimeMs?: number;
  grammarTags?: string[];
  extra?: Record<string, any>;
}

export interface SaveResult {
  sessionSaved: boolean;
  xpAwarded: number;
  srsEnrolled: number;
  streakUpdated: boolean;
  questsUpdated: boolean;
  achievementsUnlocked: any[];
  errors: string[];
}

export async function executeSavePipeline(ctx: SaveContext): Promise<SaveResult> {
  const result: SaveResult = {
    sessionSaved: false,
    xpAwarded: 0,
    srsEnrolled: 0,
    streakUpdated: false,
    questsUpdated: false,
    achievementsUnlocked: [],
    errors: [],
  };

  await StorageService.transaction(async (tx) => {
    // 1. 写 session 主表（各板块自己的 session 表）
    await saveBoardSession(tx, ctx);

    // 2. 写 response 明细
    await saveResponses(tx, ctx);

    // 3. XP 奖励
    try {
      const xpResult = await awardXP(
        ctx.userId,
        `session_${ctx.board}`,
        calculateSessionXP(ctx),
        ctx.board,
        ctx.sessionId
      );
      result.xpAwarded = xpResult.totalXP;
    } catch (e: any) {
      result.errors.push(`XP: ${e.message}`);
    }

    // 4. SRS 入队（错题 + 新学项）
    try {
      let enrolled = 0;
      for (const r of ctx.responses) {
        if (!r.correct || r.rating) {
          await enrollSRSItem(tx, ctx.userId, r.itemType, r.itemId, r.correct);
          enrolled++;
        }
      }
      result.srsEnrolled = enrolled;
    } catch (e: any) {
      result.errors.push(`SRS: ${e.message}`);
    }

    // 5. Streak 更新
    try {
      const streakResult = await updateStreak(ctx.userId);
      result.streakUpdated = streakResult.changed;
    } catch (e: any) {
      result.errors.push(`Streak: ${e.message}`);
    }

    // 6. 每日任务
    try {
      await updateDailyQuests(ctx.userId, 'session_complete');
      result.questsUpdated = true;
    } catch (e: any) {
      result.errors.push(`Quests: ${e.message}`);
    }

    // 7. 成就 / 等级检查
    try {
      const unlocked = await checkAchievements(ctx.userId, 'session_complete', ctx.correctCount);
      result.achievementsUnlocked = unlocked;
    } catch (e: any) {
      result.errors.push(`Achievements: ${e.message}`);
    }

    result.sessionSaved = true;
  });

  return result;
}

function calculateSessionXP(ctx: SaveContext): number {
  // 基础：每题正确 +2，错误标记 +1
  let xp = 0;
  for (const r of ctx.responses) {
    xp += r.correct ? 2 : 1;
  }
  // 板块完成奖励
  xp += 10;
  // 连击奖励（连续正确 ≥5）
  let consecutiveCorrect = 0;
  for (const r of ctx.responses) {
    if (r.correct) {
      consecutiveCorrect++;
    } else {
      consecutiveCorrect = 0;
    }
  }
  if (consecutiveCorrect >= 5) xp += 5;

  return xp;
}

async function enrollSRSItem(
  tx: Transaction,
  userId: string,
  itemType: string,
  itemId: string,
  wasCorrect: boolean
): Promise<void> {
  const difficulty = wasCorrect ? 5.0 : 7.0;
  const stability = wasCorrect ? 1.0 : 0.5;

  await tx.execute(
    `INSERT OR IGNORE INTO srs_cards
      (user_id, item_type, item_id, difficulty, stability, due_date, created_at, updated_at)
     VALUES (?, ?, ?, ?, ?, ?, ?, ?)`,
    [
      userId, itemType, itemId, difficulty, stability,
      new Date().toISOString(),
      new Date().toISOString(),
      new Date().toISOString(),
    ]
  );
}
```

> **注意**：`saveBoardSession` 和 `saveResponses` 是每个板块各自实现的，公共管线只定义签名，各板块文件内提供具体实现。

// boards/speaking/SpeakingSaveSession.ts

interface SpeakingResponse extends SessionResponse {
  scenarioId: string;
  recordingUri?: string;
  recordingDurationMs?: number;
  userText?: string;        // 手动输入的文本（无 ASR 时）
  promptText: string;
}

export async function saveSpeakingSession(
  userId: string,
  sessionData: {
    startedAt: Date;
    scenarioId: string;
    responses: SpeakingResponse[];
  }
): Promise<SaveResult> {
  const sessionId = `speak_${Date.now()}_${Math.random().toString(36).slice(2, 7)}`;
  const completedAt = new Date();

  // 口语板块没有"对错"，用完成度衡量
  const completedCount = sessionData.responses.filter(r => r.correct).length;
  const totalCount = sessionData.responses.length;

  const ctx: SaveContext = {
    userId,
    board: 'speaking',
    sessionId,
    startedAt: sessionData.startedAt.toISOString(),
    completedAt: completedAt.toISOString(),
    totalItems: totalCount,
    correctCount: completedCount,
    wrongCount: 0,
    skippedCount: totalCount - completedCount,
    accuracy: completedCount / totalCount || 0,
    timeSpentSec: Math.round((completedAt.getTime() - sessionData.startedAt.getTime()) / 1000),
    level: await getCurrentLevel(userId),
    responses: sessionData.responses.map(r => ({
      itemId: r.scenarioId,
      itemType: 'speaking_scenario',
      correct: r.correct,
      responseTimeMs: r.responseTimeMs,
      grammarTags: [],
      extra: {
        recordingUri: r.recordingUri,
        recordingDurationMs: r.recordingDurationMs,
        userText: r.userText,
        promptText: r.promptText,
      },
    })),
    metadata: { scenarioId: sessionData.scenarioId },
  };

  return executeSavePipeline(ctx);
}

async function saveBoardSession(tx: Transaction, ctx: SaveContext): Promise<void> {
  await tx.execute(
    `INSERT INTO speaking_sessions
      (id, user_id, scenario_id, level, total_items, completed_count,
       accuracy, time_spent_sec, started_at, completed_at, created_at)
     VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`,
    [
      ctx.sessionId, ctx.userId, ctx.metadata?.scenarioId || '',
      ctx.level, ctx.totalItems, ctx.correctCount,
      ctx.accuracy, ctx.timeSpentSec,
      ctx.startedAt, ctx.completedAt, new Date().toISOString(),
    ]
  );
}

async function saveResponses(tx: Transaction, ctx: SaveContext): Promise<void> {
  for (const r of ctx.responses) {
    // 写口语响应
    await tx.execute(
      `INSERT INTO speaking_responses
        (session_id, user_id, scenario_id, prompt_text, user_text,
         recording_path, recording_duration_ms, completed, created_at)
       VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)`,
      [
        ctx.sessionId, ctx.userId, r.itemId,
        r.extra?.promptText || '',
        r.extra?.userText || null,
        r.extra?.recordingUri || null,
        r.extra?.recordingDurationMs || null,
        r.correct ? 1 : 0,
        new Date().toISOString(),
      ]
    );

    // 如果有录音，写 speaking_recordings 表
    if (r.extra?.recordingUri) {
      await tx.execute(
        `INSERT INTO speaking_recordings
          (user_id, session_id, file_path, duration_ms, size_bytes, created_at)
         VALUES (?, ?, ?, ?, 0, ?)`,
        [
          ctx.userId, ctx.sessionId,
          r.extra.recordingUri,
          r.extra?.recordingDurationMs || 0,
          new Date().toISOString(),
        ]
      );
    }
  }
}

interface WritingResponse extends SessionResponse {
  promptId: string;
  userText: string;
  wordCount: number;
  grammarErrors?: { rule: string; message: string; position: number }[];
  aiScore?: number;  // 0-100
}

export async function saveWritingSession(
  userId: string,
  sessionData: {
    startedAt: Date;
    promptId: string;
    responses: WritingResponse[];
  }
): Promise<SaveResult> {
  const sessionId = `write_${Date.now()}_${Math.random().toString(36).slice(2, 7)}`;
  const completedAt = new Date();

  const submittedCount = sessionData.responses.filter(r => r.correct).length;

  const ctx: SaveContext = {
    userId,
    board: 'writing',
    sessionId,
    startedAt: sessionData.startedAt.toISOString(),
    completedAt: completedAt.toISOString(),
    totalItems: sessionData.responses.length,
    correctCount: submittedCount,
    wrongCount: 0,
    skippedCount: sessionData.responses.length - submittedCount,
    accuracy: submittedCount / sessionData.responses.length || 0,
    timeSpentSec: Math.round((completedAt.getTime() - sessionData.startedAt.getTime()) / 1000),
    level: await getCurrentLevel(userId),
    responses: sessionData.responses.map(r => ({
      itemId: r.promptId,
      itemType: 'writing_prompt',
      correct: r.correct,
      responseTimeMs: r.responseTimeMs,
      grammarTags: r.grammarErrors?.map(e => e.rule) || [],
      extra: {
        userText: r.userText,
        wordCount: r.wordCount,
        aiScore: r.aiScore,
        grammarErrorsJson: JSON.stringify(r.grammarErrors || []),
      },
    })),
    metadata: { promptId: sessionData.promptId },
  };

  return executeSavePipeline(ctx);
}

async function saveBoardSession(tx: Transaction, ctx: SaveContext): Promise<void> {
  await tx.execute(
    `INSERT INTO writing_sessions
      (id, user_id, prompt_id, level, total_items, submitted_count,
       avg_ai_score, time_spent_sec, started_at, completed_at, created_at)
     VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`,
    [
      ctx.sessionId, ctx.userId, ctx.metadata?.promptId || '',
      ctx.level, ctx.totalItems, ctx.correctCount,
      ctx.responses.reduce((sum, r) => sum + (r.extra?.aiScore || 0), 0) / ctx.totalItems,
      ctx.timeSpentSec,
      ctx.startedAt, ctx.completedAt, new Date().toISOString(),
    ]
  );
}

async function saveResponses(tx: Transaction, ctx: SaveContext): Promise<void> {
  for (const r of ctx.responses) {
    await tx.execute(
      `INSERT INTO writing_responses
        (session_id, user_id, prompt_id, user_text, word_count,
         ai_score, grammar_errors_json, completed, created_at)
       VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)`,
      [
        ctx.sessionId, ctx.userId, r.itemId,
        r.extra?.userText || '',
        r.extra?.wordCount || 0,
        r.extra?.aiScore || null,
        r.extra?.grammarErrorsJson || null,
        r.correct ? 1 : 0,
        new Date().toISOString(),
      ]
    );

    // 写作中标记的搭配/表达 → SRS
    if (r.correct && r.extra?.userText) {
      // 提取 collocation（简化：由 AI 在评分时标记）
      // 这里只 enroll prompt 本身
      await tx.execute(
        `INSERT OR IGNORE INTO srs_cards
          (user_id, item_type, item_id, difficulty, stability, due_date, created_at, updated_at)
         VALUES (?, 'writing_prompt', ?, 5.0, 1.0, ?, ?, ?)`,
        [ctx.userId, r.itemId, new Date().toISOString(), new Date().toISOString(), new Date().toISOString()]
      );
    }
  }
}

## 六、`.cursorrules` 追加条款

```markdown
## 板块保存规范

1. 每个板块只有一个 saveXxxSession() 入口函数
2. 所有落盘必须在 StorageService.transaction() 中
3. saveXxxSession() 内部调用 executeSavePipeline() 统一级联 XP/SRS/streak/quests
4. 板块 session 表 id 用 TEXT（格式：board_timestamp_random）
5. 板块 response 表必须有 session_id 外键关联
6. 口语板块没有"对错"，用 completed 标记完成度
7. 写作板块 grammar_errors 存 JSON，不拆子表
8. 所有 save 函数返回 SaveResult，包含级联结果
9. 重复调用同一 session 的 save 必须幂等（检查 session_id 是否已存在）
10. UI 层只调 saveXxxSession()，不直接操作数据库
```
